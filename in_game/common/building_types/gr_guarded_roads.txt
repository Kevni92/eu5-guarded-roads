# Guarded Roads
# Military road-security buildings tied directly to the four vanilla road tiers.
#
# The tiers deliberately do NOT use `obsolete`. In EU5, `obsolete` makes an
# earlier building unavailable globally once the later type is available, which
# is undesirable while a country still has older roads. Instead, the highest
# road type currently present at a Location determines which tier can be built.
# A completed higher-tier guard building removes lower-tier guard buildings in
# the same Location via `remove_if`.

# Tier I — Gravel Road / Age of Traditions
gr_guarded_gravel_roads = {
	is_foreign = no
	pop_type = soldiers
	max_levels = 1
	category = military_category
	forbidden_for_estates = yes
	employment_size = manpower_employment
	price = gr_guarded_gravel_roads_price
	build_time = soldier_building
	always_add_demands = yes

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	country_potential = {
		has_advance = road_building
	}

	# Build the Geleithaus only where Gravel Road is the highest road tier
	# currently connected to this Location.
	location_potential = {
		any_neighbor_location = {
			has_road_of_type_to = {
				target = root
				type = road_type:gravel_road
			}
		}
		NOT = {
			any_neighbor_location = {
				has_road_of_type_to = {
					target = root
					type = road_type:paved_road
				}
			}
		}
		NOT = {
			any_neighbor_location = {
				has_road_of_type_to = {
					target = root
					type = road_type:modern_road
				}
			}
		}
		NOT = {
			any_neighbor_location = {
				has_road_of_type_to = {
					target = root
					type = road_type:railroad
				}
			}
		}
	}

	allow = {
		any_neighbor_location = {
			has_road_of_type_to = {
				target = root
				type = road_type:gravel_road
			}
		}
		NOT = {
			any_neighbor_location = {
				has_road_of_type_to = {
					target = root
					type = road_type:paved_road
				}
			}
		}
		NOT = {
			any_neighbor_location = {
				has_road_of_type_to = {
					target = root
					type = road_type:modern_road
				}
			}
		}
		NOT = {
			any_neighbor_location = {
				has_road_of_type_to = {
					target = root
					type = road_type:railroad
				}
			}
		}
	}

	# Once a higher guard building has been completed here, remove this tier.
	remove_if = {
		location = {
			OR = {
				has_building = building_type:gr_guarded_paved_roads
				has_building = building_type:gr_guarded_modern_roads
				has_building = building_type:gr_guarded_rail_network
			}
		}
	}

	modifier = {
		local_distance_from_capital_speed_propagation = 0.05
	}

	raw_modifier = {
		local_manpower = -0.05
	}

	possible_production_methods = {
		gr_guarded_gravel_roads_maintenance
	}

	construction_demand = early_soldier_building_construction
}
